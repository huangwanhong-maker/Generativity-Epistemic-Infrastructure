cmake_minimum_required(VERSION 3.24)
project(GRDiagonal VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(GR 0.6.0 EXACT REQUIRED COMPONENTS data_standard)
option(GR_DIAGONAL_BUILD_TESTS "Build standalone extension conformance" ON)
add_library(gr_diagonal STATIC src/diagonal.cpp)
add_library(GRDiagonal::diagonal ALIAS gr_diagonal)
set_target_properties(gr_diagonal PROPERTIES EXPORT_NAME diagonal POSITION_INDEPENDENT_CODE ON)
target_compile_features(gr_diagonal PUBLIC cxx_std_20)
target_include_directories(gr_diagonal PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(gr_diagonal PUBLIC GR::data_standard)
file(READ manifest.json GR_DIAGONAL_MANIFEST)
configure_file(cmake/manifest.hpp.in generated/manifest.hpp @ONLY)
target_include_directories(gr_diagonal PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
add_library(gr_diagonal_native SHARED src/c_api.cpp)
target_compile_features(gr_diagonal_native PRIVATE cxx_std_20)
target_compile_definitions(gr_diagonal_native PRIVATE GR_DIAGONAL_EXPORTS)
target_link_libraries(gr_diagonal_native PRIVATE gr_diagonal)
foreach(target gr_diagonal gr_diagonal_native)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
install(TARGETS gr_diagonal gr_diagonal_native EXPORT GRDiagonalTargets RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES manifest.json README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/gr_diagonal)
configure_package_config_file(cmake/GRDiagonalConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/GRDiagonalConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/GRDiagonal)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/GRDiagonalConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(EXPORT GRDiagonalTargets NAMESPACE GRDiagonal:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/GRDiagonal)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/GRDiagonalConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/GRDiagonalConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/GRDiagonal)
if(GR_DIAGONAL_BUILD_TESTS)
  enable_testing()
  add_executable(test_diagonal tests/conformance.cpp)
  target_link_libraries(test_diagonal PRIVATE gr_diagonal)
  add_test(NAME diagonal_conformance COMMAND test_diagonal)
endif()
