cmake_minimum_required(VERSION 3.24)
project(GRRelationalAnalysis VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(GR 0.7.0 EXACT REQUIRED COMPONENTS data_relational)
option(GR_MODULE_BUILD_TESTS "Build standalone package conformance" ON)
option(GR_MODULE_BUILD_EXAMPLES "Build standalone package example" ON)
option(GR_MODULE_BUILD_BENCHMARKS "Build separate benchmark executable" OFF)
add_library(gr_relational_analysis STATIC src/analysis.cpp src/package.cpp)
add_library(GRRelationalAnalysis::analysis ALIAS gr_relational_analysis)
set_target_properties(gr_relational_analysis PROPERTIES EXPORT_NAME analysis POSITION_INDEPENDENT_CODE ON)
target_compile_features(gr_relational_analysis PUBLIC cxx_std_20)
target_include_directories(gr_relational_analysis PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(gr_relational_analysis PUBLIC GR::data_relational)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/python/src/gr_relational_analysis/manifest.json")
file(READ python/src/gr_relational_analysis/manifest.json GR_MODULE_MANIFEST)
configure_file(cmake/manifest.hpp.in generated/manifest.hpp @ONLY)
target_include_directories(gr_relational_analysis PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
add_library(gr_relational_analysis_native SHARED src/c_api.cpp)
target_compile_features(gr_relational_analysis_native PRIVATE cxx_std_20)
target_compile_definitions(gr_relational_analysis_native PRIVATE GR_MODULE_EXPORTS)
target_link_libraries(gr_relational_analysis_native PRIVATE gr_relational_analysis)
foreach(target gr_relational_analysis gr_relational_analysis_native)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
install(TARGETS gr_relational_analysis gr_relational_analysis_native EXPORT GRRelationalAnalysisTargets RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES python/src/gr_relational_analysis/manifest.json README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/gr_relational_analysis)
configure_package_config_file(cmake/GRRelationalAnalysisConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/GRRelationalAnalysisConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/GRRelationalAnalysis)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/GRRelationalAnalysisConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(EXPORT GRRelationalAnalysisTargets NAMESPACE GRRelationalAnalysis:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/GRRelationalAnalysis)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/GRRelationalAnalysisConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/GRRelationalAnalysisConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/GRRelationalAnalysis)
if(GR_MODULE_BUILD_TESTS)
  enable_testing()
  add_executable(test_analysis tests/analysis.cpp)
  target_link_libraries(test_analysis PRIVATE gr_relational_analysis)
  add_test(NAME analysis_conformance COMMAND test_analysis)
endif()
if(GR_MODULE_BUILD_EXAMPLES)
  add_executable(example_analysis examples/native.cpp)
  target_link_libraries(example_analysis PRIVATE gr_relational_analysis)
endif()
if(GR_MODULE_BUILD_BENCHMARKS)
  add_executable(benchmark_analysis benchmarks/analysis.cpp)
  target_link_libraries(benchmark_analysis PRIVATE gr_relational_analysis)
  if(WIN32)
    target_link_libraries(benchmark_analysis PRIVATE psapi)
  endif()
endif()
