cmake_minimum_required(VERSION 3.24)
project(GRVisualization VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(GR 0.7.0 EXACT REQUIRED COMPONENTS data_relational)
find_package(GRRelationalAnalysis 0.1.0 EXACT REQUIRED)
option(GR_MODULE_BUILD_TESTS "Build standalone package conformance" ON)
option(GR_MODULE_BUILD_EXAMPLES "Build standalone package example" ON)
option(GR_MODULE_BUILD_BENCHMARKS "Build separate benchmark executable" OFF)
add_library(gr_visualization STATIC src/visualization.cpp src/package.cpp)
add_library(GRVisualization::visualization ALIAS gr_visualization)
set_target_properties(gr_visualization PROPERTIES EXPORT_NAME visualization POSITION_INDEPENDENT_CODE ON)
target_compile_features(gr_visualization PUBLIC cxx_std_20)
target_include_directories(gr_visualization PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(gr_visualization PUBLIC GRRelationalAnalysis::analysis)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/python/src/gr_visualization/manifest.json")
file(READ python/src/gr_visualization/manifest.json GR_MODULE_MANIFEST)
configure_file(cmake/manifest.hpp.in generated/manifest.hpp @ONLY)
target_include_directories(gr_visualization PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
add_library(gr_visualization_native SHARED src/c_api.cpp)
target_compile_features(gr_visualization_native PRIVATE cxx_std_20)
target_compile_definitions(gr_visualization_native PRIVATE GR_MODULE_EXPORTS)
target_link_libraries(gr_visualization_native PRIVATE gr_visualization)
foreach(target gr_visualization gr_visualization_native)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
install(TARGETS gr_visualization gr_visualization_native EXPORT GRVisualizationTargets RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES python/src/gr_visualization/manifest.json README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/gr_visualization)
configure_package_config_file(cmake/GRVisualizationConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/GRVisualizationConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/GRVisualization)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/GRVisualizationConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(EXPORT GRVisualizationTargets NAMESPACE GRVisualization:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/GRVisualization)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/GRVisualizationConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/GRVisualizationConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/GRVisualization)
if(GR_MODULE_BUILD_TESTS)
  enable_testing()
  add_executable(test_visualization tests/conformance.cpp)
  target_link_libraries(test_visualization PRIVATE gr_visualization)
  add_test(NAME visualization_conformance COMMAND test_visualization)
endif()
if(GR_MODULE_BUILD_EXAMPLES)
  add_executable(example_visualization examples/demo.cpp)
  target_link_libraries(example_visualization PRIVATE gr_visualization)
endif()
if(GR_MODULE_BUILD_BENCHMARKS)
  add_executable(benchmark_visualization benchmarks/visualization.cpp)
  target_link_libraries(benchmark_visualization PRIVATE gr_visualization)
endif()
