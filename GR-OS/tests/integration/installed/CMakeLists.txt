cmake_minimum_required(VERSION 3.24)
project(GRInstalledConsumer LANGUAGES C CXX)
find_package(GR 0.7 REQUIRED COMPONENTS core storage execution runtime native buffer)
add_executable(consumer_cpp main.cpp)
target_link_libraries(consumer_cpp PRIVATE GR::runtime)
if(TARGET GR::experimental_data)
  target_link_libraries(consumer_cpp PRIVATE GR::experimental_data)
  target_compile_definitions(consumer_cpp PRIVATE GR_CONSUMER_EXPERIMENTAL)
endif()
add_executable(consumer_c main.c)
target_link_libraries(consumer_c PRIVATE GR::native)

if(TARGET GR::data)
  target_link_libraries(consumer_cpp PRIVATE GR::data GR::data_standard GR::data_numeric GR::data_sparse GR::data_relational)
  target_compile_definitions(consumer_cpp PRIVATE GR_CONSUMER_DATA)
endif()
